2. Infrastructure Layer: Entity Framework SQL Injection
Modern .NET uses Entity Framework (EF) Core, which prevents SQL injection by default. We will forcefully write a raw, concatenated query.

Navigate to the backend's Data/Infrastructure project.

Find any Repository class or create a new one: Infrastructure/Repositories/VulnerableRepository.cs.

Add this code:

using Microsoft.EntityFrameworkCore;
using System.Linq;

namespace TwitterClone.Infrastructure.Repositories;

public class VulnerableRepository
{
    private readonly DbContext _context;
    
    public VulnerableRepository(DbContext context)
    {
        _context = context;
    }

    public object GetUserBadly(string username)
    {
        // VULNERABILITY: EF Core SQL Injection via raw string concatenation (CWE-89)
        string query = $"SELECT * FROM AspNetUsers WHERE UserName = '{username}'";
        
        // Snyk will detect this because it uses FromSqlRaw instead of parameterized inputs
        return _context.Database.SqlQueryRaw<int>(query).ToList();
    }
}

using System.Net.Http;

namespace TwitterClone.Application.Services;

public class MediaProcessingService
{
    public async Task<string> DownloadProfilePictureAsync(string userProvidedUrl)
    {
        // VULNERABILITY: SSRF (Server-Side Request Forgery) (CWE-918)

        using var client = new HttpClient();

4. API Layer: Open Redirect & Insecure Deserialization
Navigate to the API project (WebAPI).

Open any Controller (or create WebAPI/Controllers/AuthController.cs).

Add these endpoints:
using Microsoft.AspNetCore.Mvc;
using Newtonsoft.Json;

namespace TwitterClone.WebAPI.Controllers;

[ApiController]
[Route("api/[controller]")]
public class AuthController : ControllerBase
{
    [HttpGet("redirect")]
    public IActionResult LoginRedirect([FromQuery] string returnUrl)
    {
        // VULNERABILITY: Open Redirect (CWE-601)
        // Attackers can trick users into clicking a link to the real API that redirects to a malicious site
        return Redirect(returnUrl);
    }

    [HttpPost("deserialize")]
    public IActionResult ProcessWebhook([FromBody] string untrustedJson)
    {
        // VULNERABILITY: Insecure Deserialization (CWE-502)
        // Using TypeNameHandling.All allows an attacker to instantiate arbitrary classes
        var settings = new JsonSerializerSettings { TypeNameHandling = TypeNameHandling.All };
        var obj = JsonConvert.DeserializeObject(untrustedJson, settings);
        
        return Ok();
    }
}
